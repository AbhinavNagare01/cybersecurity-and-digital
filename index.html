HOST = "127.0.0.1"
PORT = int(os.environ.get("LOCALSHIELD_PORT", "5050"))
ALLOWED_HOSTS = {f"127.0.0.1:{PORT}", f"localhost:{PORT}", "127.0.0.1", "localhost"}
ALLOWED_ORIGINS = {f"http://127.0.0.1:{PORT}", f"http://localhost:{PORT}", "null"}

# Common developer ports to probe rapidly on localhost and 0.0.0.0
COMMON_DEV_PORTS = {
    3000: ("TCP/HTTP", "Node.js / React / Next.js Dev"),
    3001: ("TCP/HTTP", "Node.js Secondary API"),
    4200: ("TCP/HTTP", "Angular CLI Dev Server"),
    5000: ("TCP/HTTP", "Flask / AirPlay / Local API"),

def log_event(event_type: str, message: str, color: str = "cyan") -> None:
    FIREWALL_LOGS.insert(

def parse_os_listeners():
    """Inspect OS listening sockets via ss, lsof, or netstat to detect 0.0.0.0 vs 127.0.0.1 bindings."""
    discovered = {}
    system = platform.system().lower()

    try:
        if system == "linux":
            out = subprocess.check_output(
                ["ss", "-tulnp"], stderr=subprocess.DEVNULL, text=True, timeout=2.0
            )
            for line in out.splitlines()[1:]:
                parts = line.split()
                if len(parts) < 5:
                    continue
                local_addr = parts[4]
                if ":" not in local_addr:
                    continue
                host_part, port_str = local_addr.rsplit(":", 1)
                if not port_str.isdigit():
                    continue
                port_num = int(port_str)
                bind_ip = (
                    "0.0.0.0"
                    if host_part in ("0.0.0.0", "*", "[::]", "::")
                    else "127.0.0.1"
                )
                proc_match = re.search(r'users:\(\("([^"]+)",pid=(\d+)', line)
                if proc_match:
                    service_str = f"{proc_match.group(1)} (PID {proc_match.group(2)})"
                    pid = int(proc_match.group(2))
                else:
                    service_str = COMMON_DEV_PORTS.get(
                        port_num, ("TCP", f"Local Daemon :{port_num}")
                    )[1]
                    pid = None
                discovered[port_num] = {
                    "id": f"p-{port_num}",
                    "port": port_num,
                    "proto": COMMON_DEV_PORTS.get(port_num, ("TCP", ""))[0],
                    "service": service_str,
                    "pid": pid,
                    "bind": bind_ip,
                    "auth": "Password Auth" if port_num == 5432 else "Unauthenticated",
                }
        elif system == "darwin":
            out = subprocess.check_output(
                ["lsof", "-iTCP", "-sTCP:LISTEN", "-P", "-n"],
                stderr=subprocess.DEVNULL,
                text=True,
                timeout=2.0,
            )
            for line in out.splitlines()[1:]:
                parts = line.split()
                if len(parts) < 9:
                    continue
                cmd_name, pid_str = parts[0], parts[1]
                addr = parts[8]
                if ":" not in addr:
                    continue
                host_part, port_str = addr.rsplit(":", 1)
                if not port_str.isdigit():
                    continue
                port_num = int(port_str)
                bind_ip = "0.0.0.0" if host_part in ("*", "0.0.0.0") else "127.0.0.1"
                discovered[port_num] = {
                    "id": f"p-{port_num}",
                    "port": port_num,
                    "proto": COMMON_DEV_PORTS.get(port_num, ("TCP", ""))[0],
                    "service": f"{cmd_name} (PID {pid_str})",
                    "pid": int(pid_str) if pid_str.isdigit() else None,
                    "bind": bind_ip,
                    "auth": "Unauthenticated",
                }
    except Exception:
        pass

    return discovered


def scan_local_ports():
    """Combine OS socket table with fast loopback TCP probes across common dev ports."""
    listeners = parse_os_listeners()

    for port_num, (proto, label) in COMMON_DEV_PORTS.items():
        if port_num in listeners:
            continue
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as sock:
            sock.settimeout(0.08)
            if sock.connect_ex(("127.0.0.1", port_num)) == 0:
                listeners[port_num] = {
                    "id": f"p-{port_num}",
                    "port": port_num,
                    "proto": proto,
                    "service": label,
                    "pid": None,
                    "bind": "127.0.0.1",
                    "auth": "Password Auth" if port_num == 5432 else "Unauthenticated",
                }

    ports_list = sorted(listeners.values(), key=lambda x: x["port"])
    vulnerabilities = build_vulnerabilities_from_ports(ports_list)
    open_vulns = sum(1 for v in vulnerabilities if v["status"] == "OPEN")
    exposed_ports = sum(1 for p in ports_list if p["bind"] == "0.0.0.0")
    score = max(15, min(100, 100 - (open_vulns * 6) - (exposed_ports * 8)))

    return {
        "timestamp": time.strftime("%Y-%m-%dT%H:%M:%S"),
        "host": HOST,
        "score": score,
        "exposed_count": exposed_ports,
        "total_ports": len(ports_list),
        "ports": ports_list,
        "vulnerabilities": vulnerabilities,
    }


def build_vulnerabilities_from_ports(ports_list):
    vulns = []
    for p in ports_list:
        port = p["port"]
        is_exposed = p["bind"] == "0.0.0.0"
        if port == 6379:
            vulns.append(
                {
                    "id": "vuln-6379",
                    "title": f"Redis Server Active on {p['bind']}:6379",
                    "port": 6379,
                    "severity": "CRITICAL" if is_exposed else "HIGH",
                    "status": "OPEN",
                    "desc": "Redis instance detected. Ensure protected-mode is enabled and bind is locked to 127.0.0.1.",
                    "cmd": 'redis-cli CONFIG SET bind "127.0.0.1" && redis-cli CONFIG SET protected-mode yes',
                }
            )
        elif port == 9229:
            vulns.append(
                {
                    "id": "vuln-9229",
                    "title": "Node.js V8 Inspector Debugger Listening (:9229)",
                    "port": 9229,
                    "severity": "CRITICAL",
                    "status": "OPEN",
                    "desc": "Active Node.js debugger port can allow arbitrary code execution via WebSocket or DNS rebinding.",
                    "cmd": "node --inspect=127.0.0.1:9229 server.js",
                }
            )
        elif is_exposed:
            vulns.append(
                {
                    "id": f"vuln-{port}",
                    "title": f"Service on :{port} ({p['service']}) Exposed to LAN (0.0.0.0)",
                    "port": port,
                    "severity": "HIGH",
                    "status": "OPEN",
                    "desc": f"Port :{port} is bound to 0.0.0.0 instead of 127.0.0.1, making it reachable by peers on shared Wi-Fi.",
                    "cmd": f"# Restart service on :{port} with --host 127.0.0.1",
                }
            )
    return vulns


def audit_target_headers(target_url: str):
    """Perform a safe localhost-only HTTP header & CORS check."""
    parsed = urllib.parse.urlparse(target_url)
    if parsed.scheme not in ("http", "https") or parsed.hostname not in (
        "127.0.0.1",
        "localhost",
    ):
        return {
            "error": "Target must be an http://localhost or http://127.0.0.1 URL (SSRF guard)."
        }

    req = urllib.request.Request(
        target_url,
        headers={
            "User-Agent": "LocalShield-Auditor/1.0",
            "Origin": "https://untrusted-lan-peer.example",
        },
        method="GET",
    )
    try:
        with urllib.request.urlopen(req, timeout=2.5) as resp:
            headers = {k.lower(): v for k, v in resp.getheaders()}
            status_code = resp.status
    except urllib.error.HTTPError as e:
        headers = {k.lower(): v for k, v in e.headers.items()} if e.headers else {}
        status_code = e.code
    except Exception as exc:
        return {
            "reachable": False,
            "target": target_url,
            "error": f"Could not connect to {target_url}: {exc}",
            "checks": [],
        }

    checks = []
    # 1. Server / X-Powered-By banner check
    powered = headers.get("x-powered-by") or headers.get("server")
    checks.append(
        {
            "header": "X-Powered-By / Server",
            "value": powered if powered else "Hidden",
            "status": "WARN" if powered else "PASS",
            "note": "Server stack banner exposed in response headers"
            if powered
            else "Server banner suppressed",
        }
    )

    # 2. CORS Origin check
    acao = headers.get("access-control-allow-origin", "Not Set")
    cors_bad = acao in ("*", "https://untrusted-lan-peer.example")
    checks.append(
        {
            "header": "Access-Control-Allow-Origin",
            "value": acao,
            "status": "FAIL" if cors_bad else "PASS",
            "note": "Allows arbitrary external origins to read localhost responses"
            if cors_bad
            else "Restricted origin policy",
        }
    )

    # 3. CSP check
    csp = headers.get("content-security-policy", "Missing")
    checks.append(
        {
            "header": "Content-Security-Policy",
            "value": csp,
            "status": "PASS" if csp != "Missing" else "FAIL",
            "note": "Mitigates XSS and unauthorized script execution"
            if csp != "Missing"
            else "No Content-Security-Policy header present",
        }
    )

    # 4. X-Frame-Options
    xfo = headers.get("x-frame-options", "Missing")
    checks.append(
        {
            "header": "X-Frame-Options",
            "value": xfo,
            "status": "PASS" if xfo != "Missing" else "WARN",
            "note": "Prevents iframe clickjacking on local web UIs",
        }
    )

    log_event("HEADER-AUDIT", f"Audited headers on {target_url} (HTTP {status_code})", "cyan")
    return {"reachable": True, "target": target_url, "status_code": status_code, "checks": checks}


def scan_env_content(raw_text: str):
    """Scan .env or config text for leaked credentials and unsafe localhost bindings."""
    findings = []
    sanitized_lines = []

    for idx, line in enumerate(raw_text.splitlines(), start=1):
        stripped = line.strip()
        if not stripped or stripped.startswith("#"):
            sanitized_lines.append(line)
            continue

        if re.search(r"(VITE_|NEXT_PUBLIC_|REACT_APP_)", stripped, re.I) and re.search(
            r"(SECRET|PRIVATE|KEY|TOKEN|PASSWORD)", stripped, re.I
        ):
            findings.append(
                {
                    "line": idx,
                    "severity": "CRITICAL",
                    "msg": "Secret variable exposed to client bundle via VITE_/NEXT_PUBLIC_/REACT_APP_ prefix",
                }
            )

        if re.search(
            r"(sk_live_[0-9a-zA-Z]{10,}|AKIA[0-9A-Z]{16}|ghp_[0-9a-zA-Z]{20,}|AIza[0-9A-Za-z\-_]{35})",
            stripped,
        ):
            findings.append(
                {
                    "line": idx,
                    "severity": "CRITICAL",
                    "msg": "Live cloud/API credential pattern (Stripe / AWS / GitHub / GCP) detected",
                }
            )

        if re.search(r"(0\.0\.0\.0|admin123|password123|supersecret)", stripped, re.I):
            findings.append(
                {
                    "line": idx,
                    "severity": "HIGH",
                    "msg": "Weak hardcoded secret or 0.0.0.0 LAN binding found in configuration",
                }
            )

        if "=" in line:
            key = line.split("=", 1)[0]
            clean_key = re.sub(
                r"^(VITE_|NEXT_PUBLIC_|REACT_APP_)(.*(?:SECRET|PRIVATE).*)",
                r"\2",
                key,
                flags=re.I,
            )
            sanitized_lines.append(f"{clean_key}=<REPLACE_IN_LOCAL_ENV>")
        else:
            sanitized_lines.append(line)

    log_event("SECRET-SCAN", f"Scanned config snippet ({len(findings)} issues found)", "amber" if findings else "emerald")
    return {
        "findings": findings,
        "count": len(findings),
        "sanitized": "\n".join(sanitized_lines),
    }


class LocalShieldHandler(BaseHTTPRequestHandler):
    def log_message(self, fmt, *args):
        # Keep console output clean and structured
        pass

    def _check_host_and_security(self) -> bool:
        """Enforce DNS Rebinding protection by validating the Host header."""
        host_hdr = self.headers.get("Host", "")
        if host_hdr and host_hdr not in ALLOWED_HOSTS:
            log_event(
                "BLOCKED-REBIND",
                f"Rejected request with untrusted Host header: {host_hdr}",
                "rose",
            )
            self._send_json(
                {"error": "Blocked by LocalShield DNS Rebinding Protection"}, status=403
            )
            return False
        return True

    def _send_security_headers(self, content_type: str = "application/json; charset=utf-8"):
        origin = self.headers.get("Origin", "")
        if origin in ALLOWED_ORIGINS:
            self.send_header("Access-Control-Allow-Origin", origin)
        else:
            self.send_header("Access-Control-Allow-Origin", f"http://127.0.0.1:{PORT}")
        self.send_header("Access-Control-Allow-Methods", "GET, POST, OPTIONS")
        self.send_header("Access-Control-Allow-Headers", "Content-Type")
        self.send_header("X-Frame-Options", "DENY")
        self.send_header("X-Content-Type-Options", "nosniff")
        self.send_header("Referrer-Policy", "no-referrer")
        self.send_header("Content-Type", content_type)

    def _send_json(self, payload: dict, status: int = 200):
        raw = json.dumps(payload, indent=2).encode("utf-8")
        self.send_response(status)
        self._send_security_headers("application/json; charset=utf-8")
        self.send_header("Content-Length", str(len(raw)))
        self.end_headers()
        self.wfile.write(raw)

    def do_OPTIONS(self):
        if not self._check_host_and_security():
            return
        self.send_response(204)
        self._send_security_headers()
        self.end_headers()

    def do_GET(self):
        if not self._check_host_and_security():
            return

        parsed = urllib.parse.urlparse(self.path)
        path = parsed.path

        if path in ("/", "/index.html"):
            ui_file = Path(__file__).parent / "index.html"
            if ui_file.exists():
                data = ui_file.read_bytes()
                self.send_response(200)
                self._send_security_headers("text/html; charset=utf-8")
                self.send_header("Content-Length", str(len(data)))
                self.end_headers()
                self.wfile.write(data)
            else:
                self._send_json({"status": "LocalShield Backend Running", "port": PORT})
            return

        if path == "/api/status":
            self._send_json(
                {
                    "status": "online",
                    "service": "LocalShield Python SecOps Daemon",
                    "bind_address": f"{HOST}:{PORT}",
                    "platform": platform.platform(),
                    "dns_rebinding_guard": True,
                    "pna_guard": True,
                }
            )
            return

        if path == "/api/scan-ports":
            result = scan_local_ports()
            log_event(
                "PORT-SCAN",
                f"Audited {result['total_ports']} active local ports ({result['exposed_count']} bound to 0.0.0.0)",
                "cyan",
            )
            self._send_json(result)
            return

        if path == "/api/logs":
            self._send_json({"logs": FIREWALL_LOGS})
            return

        self._send_json({"error": "Not found"}, status=404)

    def do_POST(self):
        if not self._check_host_and_security():
            return

        length = int(self.headers.get("Content-Length", "0"))
        body_raw = self.rfile.read(length).decode("utf-8") if length > 0 else "{}"
        try:
            body = json.loads(body_raw)
        except Exception:
            body = {}

        if self.path == "/api/audit-headers":
            url = body.get("url", f"http://127.0.0.1:{PORT}/api/status")
            self._send_json(audit_target_headers(url))
            return

        if self.path == "/api/scan-secrets":
            content = body.get("content", "")
            self._send_json(scan_env_content(content))
            return

        if self.path == "/api/kill-pid":
            pid = body.get("pid")
            if not isinstance(pid, int) or pid <= 1 or pid == os.getpid():
                self._send_json(
                    {"ok": False, "error": "Invalid or protected PID."}, status=400
                )
                return
            try:
                os.kill(pid, signal.SIGTERM)
                log_event("KILL-PID", f"Sent SIGTERM to local PID {pid}", "amber")
                self._send_json({"ok": True, "pid": pid, "message": f"Terminated PID {pid}"})
            except Exception as exc:
                self._send_json({"ok": False, "pid": pid, "error": str(exc)}, status=400)
            return

        self._send_json({"error": "Unknown endpoint"}, status=404)


def run_server():
    server = HTTPServer((HOST, PORT), LocalShieldHandler)
    print(f"[LocalShield] Listening strictly on http://{HOST}:{PORT} (Loopback Only)")
    print(f"[LocalShield] API Endpoints:")
    print(f"  GET  http://{HOST}:{PORT}/api/status")
    print(f"  GET  http://{HOST}:{PORT}/api/scan-ports")
    print(f"  GET  http://{HOST}:{PORT}/api/logs")
    print(f"  POST http://{HOST}:{PORT}/api/audit-headers")
    print(f"  POST http://{HOST}:{PORT}/api/scan-secrets")
    print(f"  POST http://{HOST}:{PORT}/api/kill-pid")
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        print("\n[LocalShield] Shutting down cleanly.")
        server.server_close()


if __name__ == "__main__":
    run_server()
